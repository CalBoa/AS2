def is_psd(M):
    try:
        vals = np.linalg.eigvals(M)
        return np.all(vals.real >= -1e-8)
    except:
        return False

def build_cov_matrix(sigmas, rhos):
    sig = np.array(sigmas).reshape(-1, 1)
    Sigma = sig @ sig.T * rhos
    if not is_psd(Sigma):
        Sigma += np.eye(len(sigmas)) * 1e-6
    return Sigma

def monte_carlo_frontier(mu, Sigma, n_samples=5000, long_only=True, seed=42):
    np.random.seed(seed)
    n = len(mu)
    rows = []
    for _ in range(n_samples):
        if long_only:
            w = np.random.rand(n)
            w = w / w.sum()
        else:
            w = np.random.randn(n)
            w = w / w.sum()
        mu_p = w @ mu
        sigma_p = np.sqrt(w @ Sigma @ w)
        rec = {"mu_p": mu_p, "sigma_p": sigma_p}
        for i in range(n):
            rec[f"w_{i}"] = w[i]
        rows.append(rec)
    return pd.DataFrame(rows)
